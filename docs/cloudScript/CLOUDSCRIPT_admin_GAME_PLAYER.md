## 2. Ví dụ sử dụng

```ts
const requireString = (value: unknown, fieldName: string): string => {
    if (typeof value !== "string" || value.trim().length === 0) throw new Error(fieldName + " is required");
    return value.trim();
};

const assertResponseOk = (response: CustomOperationResponse, actionName: string): void => {
    if (response.hasReturnCodeError()) throw new Error(actionName + " return-code error: " + (response.debugMessage ?? response.toString()));
    if (response.errorCode !== ErrorCode.Ok) throw new Error(actionName + " business error: " + response.errorCode);
};

// Đọc currency rồi cộng gem cho game player
handlers["grantGem"] = async (args: any, context, log) => {
    const userId = context.userId;

    const getPlayerCurrencyRequestData = new GamePlayerModels.AdminGetPlayerCurrencyRequestData();
    getPlayerCurrencyRequestData.userId = userId;
    getPlayerCurrencyRequestData.playerCurrencyKeys = ["gem"];

    const getPlayerCurrencyResponse = await admin.gamePlayer.getPlayerCurrencyAsync(getPlayerCurrencyRequestData);
    assertResponseOk(getPlayerCurrencyResponse, "getPlayerCurrencyAsync");

    const playerCurrencyParam = new GamePlayerModels.PlayerCurrencyParam();
    playerCurrencyParam.key = "gem";
    playerCurrencyParam.value = typeof args?.amount === "number" ? args.amount : 100;

    const changePlayerCurrencyRequestData = new GamePlayerModels.AdminChangePlayerCurrencyRequestData();
    changePlayerCurrencyRequestData.userId = userId;
    changePlayerCurrencyRequestData.playerCurrencies = [playerCurrencyParam];
    changePlayerCurrencyRequestData.log = "grantGem cloudscript, userId=" + userId;

    const changeResponse = await admin.gamePlayer.changePlayerCurrencyAsync(changePlayerCurrencyRequestData);
    assertResponseOk(changeResponse, "changePlayerCurrencyAsync");

    return { ok: true, granted: playerCurrencyParam.value };
};
```
